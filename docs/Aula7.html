# parcela subdividida: aplica o que tem menos níveis (neste caso, o método de inoculação). Usamos parcela subdividida porque economizamos espaço, o tamanho de área reduz pela metade. library(ggplot2)

# 1. Dados (mantendo a estrutura anterior)
df_grafico <- data.frame(
  hybrid = rep(c("30S31H", "30S31YH", "30F53 HX", "30F53 YH", "30K64", "BG7049H"), each = 2),
  method = rep(c("pin", "silk"), 6),
  media = c(38.1, 26.5, 32.5, 26.7, 25.3, 25.0, 24.6, 26.2, 20.6, 21.5, 19.4, 19.2),
  se = 3.56,
  letras = c("bA", "aA", "abA", "aA", "abA", "aA", "abA", "aA", "aA", "aA", "aA", "aA")
)

# Reordenar híbridos pela média decrescente para melhor visualização
df_grafico$hybrid <- reorder(df_grafico$hybrid, -df_grafico$media)

# 2. Gráfico com a legenda explicativa
ggplot(df_grafico, aes(x = hybrid, y = media, fill = method)) +
  geom_bar(stat = "identity", position = position_dodge(width = 0.9), color = "black") +
  geom_errorbar(aes(ymin = media - se, ymax = media + se), 
                position = position_dodge(width = 0.9), width = 0.2) +
  geom_text(aes(label = letras, y = media + se + 1.5), 
            position = position_dodge(width = 0.9), size = 3.5, fontface = "bold") +
  scale_fill_manual(values = c("pin" = "#56B4E9", "silk" = "#E69F00")) + # Cores colorblind-friendly
  labs(
    x = "Híbrido", 
    y = "Índice Médio", 
    fill = "Método",
    title = "Comparação de Índices por Híbrido e Método",
    caption = "Letras minúsculas comparam híbridos (colunas); Letras maiúsculas comparam métodos (barras lado a lado).\nMédias seguidas de mesma letra não diferem pelo teste de Tukey (p > 0,05)."
  ) +
  theme_minimal() +
  theme(
    axis.text.x = element_text(angle = 45, hjust = 1),
    plot.caption = element_text(hjust = 0, face = "italic", size = 9), # Alinha a legenda à esquerda
    legend.position = "top"
